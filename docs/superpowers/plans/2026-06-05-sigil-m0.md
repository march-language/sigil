# Sigil M0 — Skeleton Build Pipeline

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** `sigil build` reads `content/*.md`, wraps each body in `Html.layout`, writes `out/<slug>.html`. No markdown parsing, no front matter. Proves I/O + IOList + layout end to end.

**Architecture:** Three modules — `Config` (Site record + defaults), `Build` (pure pipeline), `Ssg` (CLI entry + arg parsing). Build is tested directly; CLI wires it to argv. Entry module is named `Ssg` (not `Sigil`) to avoid shadowing the stdlib's `Sigil` sigil-handler module.

**Tech Stack:** March stdlib — `File`, `Dir`, `Path`, `Html`, `IOList`, `System`, `Test`

---

mod Build do

  import Config

  pfn fmt_err(e) : String do
    to_string(e)
  end

  fn run(site : Config.Site) : Result(Unit, String) do
    match Dir.mkdir_p(site.output_dir) do
    Err(e) -> Err("cannot create output dir: " ++ fmt_err(e))
    Ok(_) ->
      match Dir.list_full(site.content_dir) do
      Err(e) -> Err("cannot list content dir: " ++ fmt_err(e))
      Ok(paths) ->
        let md_paths = List.filter(paths, fn p -> Path.extension(p) == "md")
        build_all(site, md_paths)
      end
    end
  end

  pfn build_all(site : Config.Site, paths : List(String)) : Result(Unit, String) do
    match paths do
    Nil -> Ok(())
    Cons(src, rest) ->
      match build_page(site, src) do
      Err(msg) -> Err(msg)
      Ok(_)    -> build_all(site, rest)
      end
    end
  end

  pfn build_page(site : Config.Site, src : String) : Result(Unit, String) do
    match File.read(src) do
    Err(e) -> Err("cannot read " ++ src ++ ": " ++ fmt_err(e))
    Ok(body) ->
      let slug = Path.strip_extension(Path.basename(src))
      let html = Html.layout(slug, IOList.empty(), IOList.from_string(body))
      let out  = Path.join(site.output_dir, slug ++ ".html")
      match File.write(out, IOList.to_string(html)) do
      Err(e) -> Err("cannot write " ++ out ++ ": " ++ fmt_err(e))
      Ok(_)  -> Ok(())
      end
    end
  end

end
```

mod Ssg do

  import Config
  import Build

  fn parse_args(argv : List(String)) : Result(Config.Site, String) do
    match argv do
    Nil -> Err("usage: sigil build [--content-dir DIR] [--output-dir DIR]")
    Cons(_, rest) ->
      match rest do
      Nil -> Err("usage: sigil build [--content-dir DIR] [--output-dir DIR]")
      Cons("build", flags) -> parse_flags(flags, Config.default())
      Cons(cmd, _) -> Err("unknown subcommand: " ++ cmd)
      end
    end
  end

  pfn parse_flags(args : List(String), site : Config.Site) : Result(Config.Site, String) do
    match args do
    Nil -> Ok(site)
    Cons("--content-dir", Cons(val, rest)) ->
      parse_flags(rest, { site with content_dir = val })
    Cons("--output-dir", Cons(val, rest)) ->
      parse_flags(rest, { site with output_dir = val })
    Cons(flag, _) -> Err("unknown flag: " ++ flag)
    end
  end

  fn main() do
    match parse_args(System.argv()) do
    Err(msg) -> do
      println(msg)
      System.exit(1)
    end
    Ok(site) ->
      match Build.run(site) do
      Err(msg) -> do
        println("error: " ++ msg)
        System.exit(1)
      end
      Ok(_) -> println("done.")
      end
    end
  end

end
```
